```python
TICK_SECONDS = 60
KEEP_RUNS = 10

def compute_next_run(interval_type: str, interval_value: str, from_dt: datetime | None = None) -> str:
    # daily "09:00" → 今天未到则今天，否则明天；weekly "MON,THU 09:00" → 下一个匹配星期（ISO WEEKDAYS 映射）
    # monthly "1,15 09:00" → 下一个匹配日（日超当月天数取当月最后一天）；返回 ISO 无时区字符串

class ScheduleScheduler:
    def __init__(self, db, data_dir): ...
    async def start(self): ...   # create_task(_loop)
    async def stop(self): ...    # cancel + await
    async def _loop(self):
        await self._recover_missed()
        while True:
            try:
                await self._tick()
            except Exception:
                logging.getLogger(__name__).exception("schedule tick failed")
            await asyncio.sleep(TICK_SECONDS)
    async def _recover_missed(self):
        # 启动扫描：next_run_at < now 且 enabled=1 → 逐个 _dispatch（补跑），每个间隔 30s（§5.4）
        # 过期 >24h → insert run(status='missed') + set next_run_at=compute_next_run(...)
        # 同批多条 missed 合并一条通知（detail 列订阅名清单）
    async def _tick(self):
        for s in self.db.list_due_schedules(_now_iso()):
            await self._dispatch(s)
    async def _dispatch(self, s):
        # 顺序（§5.2 + R1/R2）：
        # 1) R2 忙（_r2_busy()）→ return（deferred，next_run_at 不动）
        # 2) db.schedule_last_run_running(s.id) → insert run(skipped) + 推进 next_run_at → return
        # 3) R1：db.count_running_runs() >= 1 → return
        # 4) not deps.task_agent.can_start_task() → return（闸门满）
        # 5) task = db.create_task(title=f"{s['name']} · {M/D}", data_dir, mode='general', schedule_id=s['id'])
        #    run_id = f"run_{uuid4().hex[:8]}"; db.insert_schedule_run(run_id, s.id, task.id)
        #    db.set_schedule_state(s.id, next_run_at=compute_next_run(...), last_run_at=now)
        # 6) create_task(self._execute(s, task_id, run_id))
    def _r2_busy(self) -> bool:
        # 人工任务：agent._running_task_ids 中存在 schedule_id IS NULL 的任务（db.get_task 逐一查）
        # 模型下载：db 查 model_downloads 表 status='downloading' 存在
    async def _execute(self, s, task_id, run_id, prompt):
        agent = deps.task_agent
        # 增量注入（§6）：上期 done 报告副本 → 本期 workspace/上一期报告.md + prompt 附加增量指引
        prev = self.db.last_done_run_report(s['id'])
        incr_note = ""
        if prev and Path(prev).exists():
            shutil.copy(prev, Path(self._workspace_of(task_id)) / "上一期报告.md")
            incr_note = "工作区有《上一期报告.md》：重点报告与其相比的新增动态，延续内容简要带过。\n"
        prompt_full = incr_note + prompt + "\n\n" + OUTPUT_CONTRACT   # §8.3 产出结构约定逐字（创建时已拼接存库，此处不重复拼接——若创建时已拼，仅前缀 incr_note）
        if not agent.try_acquire_task(task_id):
            self.db.update_schedule_run(run_id, 'failed'); self._notify_fail(s, '闸门被占'); return
        try:
            self.db.save_task_message(task_id, 'user', prompt_full)
            async for _ev in agent.run(task_id, prompt_full):
                pass                                   # 事件落库在 agent 内部；SSE 无人消费
            ok = self._report_exists(task_id)          # workspace/report.md 存在且非空
        except Exception as e:
            ok = False; err = str(e)
        self._on_finish(s, task_id, run_id, ok)
    def _on_finish(self, s, task_id, run_id, ok, error=None):
        # done：副本 data/schedules/{run_id}.md（copy workspace/report.md）→ 解析摘要（## 摘要 节列表项）
        #      → update_schedule_run('done', report_path, summary) → consecutive_failures=0
        #      → 通知中心：activity_events.complete_schedule_run(db, run_id, s['name'])（Task 4 前置：activity_events.py 新增）
        #      → target 绑定且 auto_import=1 → _merge_archive（Task 10 接线，先留占位 pass，Task 10 落地）
        #      → 保留期清理（Task 10 接线）
        # failed：consecutive_failures+1 → >=3 → enabled=0 + Toast 级通知 fail_schedule_run（kind='failed' 前端必弹）
        #        → else 仅通知中心记录；next_run_at 保留不推进（恢复后从下周期继续——注意触发时已推进过，此处无需再动）
```
